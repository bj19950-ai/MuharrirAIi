package com.example.export.domain

import com.example.export.model.DocumentPageModel
import com.example.export.model.ExportDocumentModel
import com.example.export.model.WordDocumentPlan
import com.example.export.model.WordExportMode

import com.example.review.model.DiscrepancyCategory
import com.example.review.model.ReviewItemModel
import com.example.review.model.ReviewStatus

/**
 * Pure Kotlin assembler that orders pages by pageNumber and items by reading order,
 * selecting userText for approved/manually-edited items, ocrText when no review was needed,
 * and flagging unresolved/pending/stale items for highlighting in Word.
 */
object WordTextAssembler {

    fun buildPlan(
        document: ExportDocumentModel,
        pages: List<DocumentPageModel>,
        reviewItems: List<ReviewItemModel>,
        mode: WordExportMode
    ): WordDocumentPlan {
        val sortedPages = pages.sortedBy { it.pageNumber }
        val itemsByPage = reviewItems
            .groupBy { it.pageNumber }
            .mapValues { (_, items) ->
                items.sortedWith(
                    compareBy<ReviewItemModel>(
                        { it.orderIndex },
                        { it.bbox?.top ?: Float.MAX_VALUE },
                        // RTL reading order fallback within same line: right-to-left (larger right first)
                        { -(it.bbox?.right ?: 0f) },
                        { it.id }
                    )
                )
            }

        val sections = sortedPages.map { page ->
            val pageItems = itemsByPage[page.pageNumber].orEmpty()
            val segments = if (pageItems.isEmpty()) {
                buildSegmentsFromRawPageOcr(page.ocrText)
            } else {
                pageItems.mapNotNull { item -> resolveItemSegment(item) }
            }
            WordPageSection(
                pageNumber = page.pageNumber,
                segments = segments
            )
        }

        return WordDocumentPlan(
            documentTitle = document.title.ifBlank { "مستند مراجع" },
            mode = mode,
            pages = sections
        )
    }

    private fun buildSegmentsFromRawPageOcr(rawOcrText: String): List<WordTextSegment> {
        if (rawOcrText.isBlank()) return emptyList()
        return rawOcrText
            .split(Regex("\\r?\\n"))
            .map { it.trim() }
            .filter { it.isNotEmpty() }
            .map { line ->
                WordTextSegment(
                    text = line,
                    isUnresolvedHighlight = false
                )
            }
    }

    private fun resolveItemSegment(item: ReviewItemModel): WordTextSegment? {
        return when {
            item.isStale ||
                item.reviewStatus == ReviewStatus.UNRESOLVED ||
                item.reviewStatus == ReviewStatus.PENDING ||
                item.reviewStatus == ReviewStatus.STALE -> {
                val unresolvedText = item.ocrText.ifBlank { item.suggestedText.orEmpty() }.trim()
                if (unresolvedText.isEmpty()) null
                else WordTextSegment(
                    text = unresolvedText,
                    isUnresolvedHighlight = true
                )
            }

            item.reviewStatus == ReviewStatus.APPROVED_GEMINI ||
                item.reviewStatus == ReviewStatus.MANUALLY_EDITED -> {
                val approvedText = item.userText?.takeIf { it.isNotBlank() }
                    ?: item.suggestedText?.takeIf { it.isNotBlank() }
                    ?: item.ocrText
                val trimmed = approvedText.trim()
                if (trimmed.isEmpty()) null
                else WordTextSegment(
                    text = trimmed,
                    isUnresolvedHighlight = false
                )
            }

            // Category A / no review needed: use userText if present, otherwise original OCR text.
            item.category == DiscrepancyCategory.A -> {
                val finalText = (item.userText?.takeIf { it.isNotBlank() } ?: item.ocrText).trim()
                if (finalText.isEmpty()) null
                else WordTextSegment(
                    text = finalText,
                    isUnresolvedHighlight = false
                )
            }

            else -> {
                val fallback = item.ocrText.trim()
                if (fallback.isEmpty()) null
                else WordTextSegment(text = fallback, isUnresolvedHighlight = false)
            }
        }
    }
}

package com.example.export.docx

import android.content.Context
import android.os.Environment
import com.example.export.domain.WordTextAssembler
import com.example.export.model.DocumentPageModel
import com.example.export.model.ExportDocumentModel
import com.example.export.model.WordDocumentPlan
import com.example.export.model.WordExportMode
import com.example.export.model.WordTextSegment
import com.example.review.model.ReviewItemModel
import org.apache.poi.xwpf.usermodel.ParagraphAlignment
import org.apache.poi.xwpf.usermodel.XWPFDocument
import org.apache.poi.xwpf.usermodel.XWPFParagraph
import org.apache.poi.xwpf.usermodel.XWPFRun
import org.apache.xmlbeans.XmlObject
import java.io.File
import java.io.FileOutputStream
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class WordGenerator(
    private val arabicFontName: String = DEFAULT_ARABIC_FONT
) {

    fun generate(
        document: ExportDocumentModel,
        pages: List<DocumentPageModel>,
        reviewItems: List<ReviewItemModel>,
        mode: WordExportMode,
        outputFile: File
    ): File {
        require(document.reviewApprovedAt != null) {
            "لا يمكن إنشاء ملف Word قبل اعتماد المراجعة."
        }

        val plan = WordTextAssembler.buildPlan(
            document = document,
            pages = pages,
            reviewItems = reviewItems,
            mode = mode
        )

        outputFile.parentFile?.mkdirs()

        val xwpfDocument = XWPFDocument()
        try {
            enforceDocumentRtlSection(xwpfDocument)
            writePlanToDocument(xwpfDocument, plan)
            FileOutputStream(outputFile).use { fos ->
                xwpfDocument.write(fos)
            }
        } finally {
            xwpfDocument.close()
        }

        return outputFile
    }

    fun buildOutputFile(context: Context, documentTitle: String, nowMillis: Long = System.currentTimeMillis()): File {
        val documentsDir = context.getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS)
            ?: File(context.filesDir, "Documents")
        if (!documentsDir.exists()) {
            documentsDir.mkdirs()
        }
        val safeName = sanitizeFileName(documentTitle)
        val timestamp = SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(Date(nowMillis))
        return File(documentsDir, "${safeName}_$timestamp.docx")
    }

    private fun writePlanToDocument(xwpfDocument: XWPFDocument, plan: WordDocumentPlan) {
        when (plan.mode) {
            WordExportMode.MATCH_ORIGINAL -> writeMatchOriginalMode(xwpfDocument, plan)
            WordExportMode.ACADEMIC -> writeAcademicMode(xwpfDocument, plan)
        }
    }

    private fun writeMatchOriginalMode(xwpfDocument: XWPFDocument, plan: WordDocumentPlan) {
        plan.pages.forEachIndexed { pageIdx, page ->
            if (page.segments.isEmpty()) {
                val emptyParagraph = createRtlParagraph(xwpfDocument)
                val run = createStyledRun(
                    paragraph = emptyParagraph,
                    text = "",
                    fontSize = 12,
                    isBold = false,
                    isUnresolvedHighlight = false
                )
                run.setText("")
            } else {
                page.segments.forEach { segment ->
                    val paragraph = createRtlParagraph(xwpfDocument)
                    paragraph.spacingAfter = 80
                    createStyledRun(
                        paragraph = paragraph,
                        text = segment.text,
                        fontSize = 12,
                        isBold = false,
                        isUnresolvedHighlight = segment.isUnresolvedHighlight
                    )
                }
            }

            // Minimal page separation matching original document flow
            if (pageIdx < plan.pages.lastIndex) {
                val spacer = createRtlParagraph(xwpfDocument)
                spacer.spacingAfter = 160
            }
        }
    }

    private fun writeAcademicMode(xwpfDocument: XWPFDocument, plan: WordDocumentPlan) {
        // Document title heading
        val titleParagraph = createRtlParagraph(xwpfDocument).apply {
            spacingBefore = 120
            spacingAfter = 240
        }
        createStyledRun(
            paragraph = titleParagraph,
            text = plan.documentTitle,
            fontSize = 20,
            isBold = true,
            isUnresolvedHighlight = false
        )

        plan.pages.forEach { page ->
            val headingParagraph = createRtlParagraph(xwpfDocument).apply {
                spacingBefore = 200
                spacingAfter = 120
            }
            createStyledRun(
                paragraph = headingParagraph,
                text = "الصفحة ${page.pageNumber}",
                fontSize = 15,
                isBold = true,
                isUnresolvedHighlight = false
            )

            page.segments.forEach { segment ->
                writeAcademicSegmentParagraph(xwpfDocument, segment)
            }
        }
    }

    private fun writeAcademicSegmentParagraph(
        xwpfDocument: XWPFDocument,
        segment: WordTextSegment
    ) {
        val paragraph = createRtlParagraph(xwpfDocument).apply {
            spacingBefore = 60
            spacingAfter = 160
            setSpacingBetween(1.5)
        }
        createStyledRun(
            paragraph = paragraph,
            text = segment.text,
            fontSize = 14,
            isBold = false,
            isUnresolvedHighlight = segment.isUnresolvedHighlight
        )
    }

    private fun createRtlParagraph(xwpfDocument: XWPFDocument): XWPFParagraph {
        val paragraph = xwpfDocument.createParagraph()
        paragraph.alignment = ParagraphAlignment.RIGHT
        val pPr = paragraph.ctp.pPr ?: paragraph.ctp.addNewPPr()
        appendWmlElement(pPr, "bidi")
        return paragraph
    }

    private fun createStyledRun(
        paragraph: XWPFParagraph,
        text: String,
        fontSize: Int,
        isBold: Boolean,
        isUnresolvedHighlight: Boolean
    ): XWPFRun {
        val run = paragraph.createRun()
        run.setText(text)
        run.fontSize = fontSize
        run.isBold = isBold
        run.setFontFamily(arabicFontName, XWPFRun.FontCharRange.ascii)
        run.setFontFamily(arabicFontName, XWPFRun.FontCharRange.hAnsi)
        run.setFontFamily(arabicFontName, XWPFRun.FontCharRange.cs)

        val rPr = run.ctr.rPr ?: run.ctr.addNewRPr()
        if (isUnresolvedHighlight) {
            // Dual visual marker in Word: yellow highlight + dark red text color
            run.color = UNRESOLVED_TEXT_RED_HEX
            appendWmlElementWithAttr(
                parent = rPr,
                localName = "highlight",
                attrName = "val",
                attrValue = "yellow"
            )
        }
        appendWmlElement(rPr, "rtl")
        return run
    }

    private fun enforceDocumentRtlSection(xwpfDocument: XWPFDocument) {
        val body = xwpfDocument.document.body
        val sectPr = body.sectPr ?: body.addNewSectPr()
        appendWmlElement(sectPr, "bidi")
    }

    private fun appendWmlElement(parent: XmlObject, localName: String) {
        val cursor = parent.newCursor()
        try {
            cursor.toEndToken()
            cursor.beginElement(localName, WML_NAMESPACE)
        } finally {
            cursor.dispose()
        }
    }

    private fun appendWmlElementWithAttr(
        parent: XmlObject,
        localName: String,
        attrName: String,
        attrValue: String
    ) {
        val cursor = parent.newCursor()
        try {
            cursor.toEndToken()
            cursor.beginElement(localName, WML_NAMESPACE)
            cursor.insertAttributeWithValue(attrName, WML_NAMESPACE, attrValue)
        } finally {
            cursor.dispose()
        }
    }

    private fun sanitizeFileName(rawTitle: String): String {
        val cleaned = rawTitle
            .trim()
            .replace(Regex("[\\\\/:*?\"<>|]"), "_")
            .replace(Regex("\\s+"), "_")
        return cleaned.ifBlank { "exported_document" }.take(48)
    }

    companion object {
        const val DEFAULT_ARABIC_FONT = "Traditional Arabic"
        const val UNRESOLVED_TEXT_RED_HEX = "B71C1C"
        private const val WML_NAMESPACE = "http://schemas.openxmlformats.org/wordprocessingml/2006/main"
    }
}

data class WordExportUiState(
    val isModeDialogVisible: Boolean = false,
    val isExporting: Boolean = false,
    val lastExportedFile: File? = null,
    val errorMessage: String? = null
)

sealed interface WordExportEvent {
    data class ShareDocxFile(val file: File, val documentTitle: String) : WordExportEvent
}

class WordExportViewModel(
    private val dataGateway: WordExportDataGateway,
    private val wordGenerator: WordGenerator = WordGenerator(),
    private val ioDispatcher: CoroutineDispatcher = Dispatchers.IO
) : ViewModel() {

    private val _uiState = MutableStateFlow(WordExportUiState())
    val uiState: StateFlow<WordExportUiState> = _uiState.asStateFlow()

    private val _events = MutableSharedFlow<WordExportEvent>()
    val events: SharedFlow<WordExportEvent> = _events.asSharedFlow()

    fun openModeChooser() {
        _uiState.update { it.copy(isModeDialogVisible = true, errorMessage = null) }
    }

    fun dismissModeChooser() {
        _uiState.update { it.copy(isModeDialogVisible = false) }
    }

    fun dismissError() {
        _uiState.update { it.copy(errorMessage = null) }
    }

    fun exportDocument(
        context: Context,
        documentId: Long,
        mode: WordExportMode
    ) {
        val appContext = context.applicationContext
        _uiState.update {
            it.copy(
                isModeDialogVisible = false,
                isExporting = true,
                errorMessage = null
            )
        }

        viewModelScope.launch {
            try {
                val document = withContext(ioDispatcher) { dataGateway.getDocument(documentId) }
                if (document == null) {
                    _uiState.update {
                        it.copy(isExporting = false, errorMessage = "تعذّر العثور على المستند.")
                    }
                    return@launch
                }
                if (document.reviewApprovedAt == null) {
                    _uiState.update {
                        it.copy(
                            isExporting = false,
                            errorMessage = "يجب تأكيد واعتماد المراجعة أولاً قبل التصدير إلى Word."
                        )
                    }
                    return@launch
                }

                val generatedFile = withContext(ioDispatcher) {
                    val pages = dataGateway.getPagesInOrder(documentId)
                    val items = dataGateway.getReviewItemsInOrder(documentId)
                    val outFile = wordGenerator.buildOutputFile(appContext, document.title)
                    wordGenerator.generate(
                        document = document,
                        pages = pages,
                        reviewItems = items,
                        mode = mode,
                        outputFile = outFile
                    )
                }

                _uiState.update {
                    it.copy(
                        isExporting = false,
                        lastExportedFile = generatedFile,
                        errorMessage = null
                    )
                }
                _events.emit(WordExportEvent.ShareDocxFile(generatedFile, document.title))
            } catch (ce: CancellationException) {
                throw ce
            } catch (_: Exception) {
                _uiState.update {
                    it.copy(
                        isExporting = false,
                        errorMessage = "تعذّر إنشاء ملف Word. حاول مرة أخرى."
                    )
                }
            }
        }
    }
}

object WordShareLauncher {
    private const val DOCX_MIME_TYPE =
        "application/vnd.openxmlformats-officedocument.wordprocessingml.document"

    fun shareDocx(context: Context, file: File, documentTitle: String) {
        val authority = "${context.packageName}.fileprovider"
        val contentUri: Uri = FileProvider.getUriForFile(context, authority, file)

        val shareIntent = Intent(Intent.ACTION_SEND).apply {
            type = DOCX_MIME_TYPE
            putExtra(Intent.EXTRA_STREAM, contentUri)
            putExtra(Intent.EXTRA_SUBJECT, documentTitle)
            clipData = ClipData.newRawUri(documentTitle, contentUri)
            addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
        }

        val chooser = Intent.createChooser(shareIntent, "مشاركة أو فتح ملف Word")
        context.startActivity(chooser)
    }
}

How RTL Was Enforced in Word
Section level: <w:bidi/> added to sectPr.
Paragraph level: paragraph.alignment = ParagraphAlignment.RIGHT plus <w:bidi/> added to paragraph.ctp.pPr.
Run level: <w:rtl/> added to run.ctr.rPr, with Traditional Arabic assigned across FontCharRange.ascii, FontCharRange.hAnsi, and FontCharRange.cs (Complex Script).
How Unresolved Items Were Highlighted
Any item with UNRESOLVED, PENDING, or STALE status (including isStale == true) receives a yellow background highlight (<w:highlight w:val="yellow"/> on rPr) paired with dark red text (run.color = "B71C1C") so it stands out immediately in both Microsoft Word and Google Docs.
How the Share Intent Is Triggered
The .docx file is written to context.getExternalFilesDir(Environment.DIRECTORY_DOCUMENTS).
WordShareLauncher.shareDocx resolves a content:// URI via FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file) and launches Intent.createChooser wrapping Intent(Intent.ACTION_SEND) with MIME type application/vnd.openxmlformats-officedocument.wordprocessingml.document, ClipData.newRawUri, and Intent.FLAG_GRANT_READ_URI_PERMISSION.
Honest Limitations
No local compile/test run here: Modules 01–05 and the Android SDK are not present in this environment, so WordGenerator and WordGeneratorTest were not compiled or executed locally; you will verify them on your machine.
POI Android 3.17 footprint & schema pruning: com.github.SUPERCILEX.poi-android:poi:3.17 uses a shrunk OOXML schema jar (hence using XmlCursor for <w:bidi/>, <w:rtl/>, and <w:highlight/> to avoid missing XMLBeans helper classes) and increases APK size; if a runtime XML parser conflict occurs with another dependency, verify aalto-xml exclusions.
Font availability: Traditional Arabic renders natively in Microsoft Word on Windows/macOS; mobile viewers without that font installed will substitute their default system Arabic font.
In-place span replacement vs. item blocks: If a page has ReviewItemEntity records covering only part of a page's text rather than full page blocks, WordTextAssembler outputs the page's ordered review items; if PageEntity stores character offsets for inline replacement in your Module 05 schema, adapt WordTextAssembler to splice those spans into page.ocrText.
